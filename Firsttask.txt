sheet = workbook.Sheets(1)

# Find last row and column
last_row = sheet.Cells(sheet.Rows.Count, 1).End(-4162).Row  # xlUp
last_col = sheet.Cells(1, sheet.Columns.Count).End(-4159).Column  # xlToLeft

# Define range for pivot table source
source_range = sheet.Range(sheet.Cells(1, 1), sheet.Cells(last_row, last_col))

# Add a new worksheet for the pivot table
pivot_sheet = workbook.Sheets.Add()
pivot_sheet.Name = "PivotTableSheet"

# Define pivot table cache
pivot_cache = workbook.PivotCaches().Create(1, source_range)  # xlDatabase

# Create Pivot Table

# Configure Pivot Table fields (adjust based on your CSV structure)

workbook.SaveAs(r"C:\path\to\your\output.xlsx", FileFormat=51)  # xlOpenXMLWorkbook

# Close Excel
workbook.Close(True)

import win32com.client as win32

# Open Excel
excel = win32.Dispatch("Excel.Application")
excel.Visible = True  # Set to False to run in the background

# Open the CSV file (Replace with your file path)
csv_file_path = r"C:\path\to\your\file.csv"
workbook = excel.Workbooks.Open(csv_file_path)

# Get the active sheet (CSV opens as a single sheet)
sheet = workbook.Sheets(1)

# Find last row and column dynamically
last_row = sheet.Cells(sheet.Rows.Count, 1).End(-4162).Row  # xlUp
last_col = sheet.Cells(1, sheet.Columns.Count).End(-4159).Column  # xlToLeft

# Define the data range
data_range = sheet.Range(sheet.Cells(1, 1), sheet.Cells(last_row, last_col))

# Add a new worksheet for the Pivot Table
pivot_sheet = workbook.Sheets.Add()
pivot_sheet.Name = "PivotTableSheet"

# Create Pivot Cache
pivot_cache = workbook.PivotCaches().Create(1, data_range)  # xlDatabase

# Create Pivot Table
pivot_table = pivot_cache.CreatePivotTable(pivot_sheet.Cells(1, 1), "PivotTable1")

# Set "Volume" as Row Field (Grouping)
pivot_table.PivotFields("Volume").Orientation = 1  # xlRowField

# Columns to SUM
sum_columns = ["Sales", "Profit"]

# Columns to AVERAGE
avg_columns = ["Discount", "Margin"]

# Apply SUM
for col in sum_columns:
    pivot_table.AddDataField(pivot_table.PivotFields(col), f"Total {col}", -4157)  # xlSum

# Apply AVERAGE
for col in avg_columns:
    pivot_table.AddDataField(pivot_table.PivotFields(col), f"Avg {col}", -4106)  # xlAverage

# Save as Excel file
output_file = r"C:\path\to\your\output.xlsx"
workbook.SaveAs(output_file, FileFormat=51)  # xlOpenXMLWorkbook

# Close Excel
workbook.Close(True)
excel.Quit()

print(f"Pivot table created with SUM & AVERAGE for specified columns in {output_file}")
