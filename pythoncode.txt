def main():
    # Get environment variables
    s3_bucket = os.environ.get("S3_BUCKET")
    s3_key = os.environ.get("S3_KEY")  # S3 key of the CSV file
    output_s3_key = os.environ.get("OUTPUT_S3_KEY") # S3 key for the output Excel file

    if not all([s3_bucket, s3_key, output_s3_key]):
        logger.error("Missing required environment variables: S3_BUCKET, S3_KEY, OUTPUT_S3_KEY")
        return

    # Download CSV file from S3
    try:
        temp_dir = "/tmp" # Use /tmp directory for Lambda
        os.makedirs(temp_dir, exist_ok=True)
        csv_file_path = os.path.join(temp_dir, os.path.basename(s3_key))
        s3.download_file(s3_bucket, s3_key, csv_file_path)
        logger.info(f"Downloaded CSV file from S3: {s3_key}")

    except ClientError as e:
        logger.error(f"Error downloading from S3: {e.response['Error']['Message']}")
        return

    # Create pivot table
    output_file_path = os.path.join(temp_dir, os.path.basename(output_s3_key))
    pivot_file_path = create_pivot_table(csv_file_path, output_file_path)

    if pivot_file_path:
        # Upload pivot table to S3
        try:
            s3.upload_file(pivot_file_path, s3_bucket, output_s3_key)
            logger.info(f"Uploaded pivot table to S3: {output_s3_key}")
        except ClientError as e:
            logger.error(f"Error uploading to S3: {e.response['Error']['Message']}")


if __name__ == "__main__":
    main()
